cmake_minimum_required(VERSION 3.22)
project(frame_volume_mixer VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MIXER_NATIVE "Build Linux PulseAudio/OpenVR dashboard executable" ${UNIX})
include(CTest)
add_library(mixer_core src/model.cpp)
target_include_directories(mixer_core PUBLIC src)
target_compile_options(mixer_core PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING)
  add_executable(model_tests tests/model_tests.cpp)
  target_link_libraries(model_tests PRIVATE mixer_core)
  add_test(NAME model COMMAND model_tests)
endif()
if(MIXER_NATIVE)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Native application targets Linux; use -DMIXER_NATIVE=OFF on Windows")
  endif()
  find_package(PkgConfig REQUIRED)
  find_package(Threads REQUIRED)
  find_package(Vulkan REQUIRED)
  pkg_check_modules(PULSE REQUIRED IMPORTED_TARGET libpulse)
  pkg_check_modules(CAIRO REQUIRED IMPORTED_TARGET cairo)
  include(cmake/OpenVR.cmake)
  add_library(mixer_pulse src/pulse_backend.cpp)
  target_link_libraries(mixer_pulse PUBLIC mixer_core PkgConfig::PULSE Threads::Threads)
  target_compile_options(mixer_pulse PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(frame-volume-mixer src/main.cpp src/instance.cpp src/panel.cpp src/dashboard.cpp third_party/frame-mic-tuner/vk_texture.cpp)
  target_include_directories(frame-volume-mixer PRIVATE third_party/frame-mic-tuner)
  target_link_libraries(frame-volume-mixer PRIVATE mixer_pulse PkgConfig::CAIRO Vulkan::Vulkan openvr_api ${CMAKE_DL_LIBS})
  target_compile_options(frame-volume-mixer PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_definitions(frame-volume-mixer PRIVATE MIXER_VERSION="${PROJECT_VERSION}")
  if(BUILD_TESTING)
    add_executable(pulse_tests tests/pulse_tests.cpp)
    pkg_check_modules(PULSE_SIMPLE REQUIRED IMPORTED_TARGET libpulse-simple)
    target_link_libraries(pulse_tests PRIVATE mixer_pulse PkgConfig::PULSE_SIMPLE)
    add_executable(instance_tests tests/instance_tests.cpp src/instance.cpp)
    target_link_libraries(instance_tests PRIVATE mixer_core)
    add_test(NAME instance COMMAND instance_tests)
  endif()
  install(TARGETS frame-volume-mixer RUNTIME DESTINATION .)
  install(DIRECTORY packaging/ DESTINATION . FILES_MATCHING PATTERN "*.sh" PATTERN "*.png" PATTERN "*.vrmanifest")
  install(FILES third_party/frame-mic-tuner/LICENSE DESTINATION licenses/frame-mic-tuner)
  install(FILES README.md design.md THIRD_PARTY.md DESTINATION .)
  install(FILES docs/validation.md docs/frame-audio-duplicates.md DESTINATION docs)
  install(FILES "${openvr_SOURCE_DIR}/LICENSE" DESTINATION licenses/openvr)
endif()
